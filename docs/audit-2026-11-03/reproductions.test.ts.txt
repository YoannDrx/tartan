import { randomUUID } from "node:crypto";
import type Stripe from "stripe";
import { afterEach, expect, it, vi } from "vitest";
import { prisma } from "@/lib/prisma";
import { handleStripeEvent } from "@/server/billing.service";
import { dispatchPending, registerDevice } from "@/server/notify.service";
import { makeOrg, makeUser } from "./helpers";
import { buildExerciseInput } from "../../mobile/src/lib/exercise-form";
import { listExercises, saveExercise } from "@/server/exercise.service";

vi.mock("web-push", () => ({ default: {
  setVapidDetails: vi.fn(),
  sendNotification: vi.fn().mockRejectedValue({ statusCode: 503 }),
} }));

afterEach(() => vi.unstubAllGlobals());

it("audit: le critère de premier démarrage doit être vrai dans un espace neuf", async () => {
  const org = await makeOrg("AuditOnboarding");
  const exercises = await listExercises(org.owner.userId, org.organizationId);
  const athletes = await prisma.athlete.count({ where: { organizationId: org.organizationId } });
  expect(athletes === 0 && exercises.every(e => e.shared)).toBe(true);
});

it("audit: renommer sur mobile conserve le média Instagram existant", async () => {
  const org = await makeOrg("AuditInstagram");
  const saved = await saveExercise(org.owner.userId, org.organizationId, {
    name: "Exercice initial", category: "plyo", tags: [], cues: [], defaultPrescription: {},
    media: { kind: "instagram", url: "https://www.instagram.com/reel/ABCDE12345/" },
  });
  const built = buildExerciseInput({ id: saved.id, name: "Exercice renommé", category: "plyo", tags: [], cues: "", base: {},
    numbers: { sets: "", reps: "", distance_m: "", duration_s: "", rest_s: "", ground_contacts: "" },
    url: "", title: "", startS: "", endS: "",
  });
  if (!built.ok) throw new Error("invalid audit fixture");
  await saveExercise(org.owner.userId, org.organizationId, built.input);
  expect((await listExercises(org.owner.userId, org.organizationId)).find(e => e.id === saved.id)?.media?.kind).toBe("instagram");
});

it("audit: un ancien événement actif ne doit pas réactiver un abonnement résilié", async () => {
  process.env.STRIPE_PRICE_COACH_MONTH = "price_audit_coach";
  process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY = "audit-public";
  process.env.VAPID_PRIVATE_KEY = "audit-private";
  const org = await makeOrg("AuditBilling");
  const event = (status: string, created: number) => ({
    id: randomUUID(), created, type: "customer.subscription.updated",
    data: { object: { id: "sub_audit", status, metadata: { organizationId: org.organizationId },
      items: { data: [{ price: { id: "price_audit_coach" }, current_period_end: 1893456000 }] } } },
  }) as unknown as Stripe.Event;
  await handleStripeEvent(event("canceled", 200));
  await handleStripeEvent(event("active", 100));
  expect((await prisma.organization.findUniqueOrThrow({ where: { id: org.organizationId } })).plan).toBe("discovery");
});

it("audit: un push web en erreur 503 doit rester en attente", async () => {
  const user = await makeUser("AuditWebPush");
  await registerDevice(user.userId, JSON.stringify({ endpoint: "https://audit.invalid/push" }), "web");
  const n = await prisma.notification.create({ data: { userId: user.userId, type: "session_reminder", payload: {} } });
  await dispatchPending(new Date());
  expect((await prisma.notification.findUniqueOrThrow({ where: { id: n.id } })).sentAt).toBeNull();
});

it("audit: une erreur de ticket Expo dans une réponse HTTP 200 doit rester en attente", async () => {
  const user = await makeUser("AuditExpo");
  await registerDevice(user.userId, "ExponentPushToken[audit]", "ios");
  vi.stubGlobal("fetch", vi.fn().mockResolvedValue(new Response(JSON.stringify({ data: [{ status: "error", details: { error: "MessageRateExceeded" } }] }), { status: 200 })));
  const n = await prisma.notification.create({ data: { userId: user.userId, type: "session_reminder", payload: {} } });
  await dispatchPending(new Date());
  expect((await prisma.notification.findUniqueOrThrow({ where: { id: n.id } })).sentAt).toBeNull();
});
